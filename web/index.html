Seção 4 · PIM IV

Arquitetura de banco de dados — SQL e NoSQL

Persistência poliglota: o relacional garante a integridade das ordens de serviço; o documental absorve a telemetria que chega a cada 5 segundos.

1FNValores atômicos: status do elevador e da rampa em colunas separadas, sem listas.
2FNChaves simples: todo atributo depende da chave inteira.
3FNNome do mecânico fica em TB_MECANICO, não repetido em cada OS.
CREATE TABLE TB_VEICULO (
    ID_VEICULO      INT AUTO_INCREMENT PRIMARY KEY,
    PLACA           VARCHAR(8)  NOT NULL UNIQUE,
    PREFIXO_IOT     VARCHAR(12) NOT NULL UNIQUE,
    MODELO          VARCHAR(50) NOT NULL,
    ANO_FABRICACAO  INT         NOT NULL,
    STATUS_ELEVADOR VARCHAR(20) DEFAULT 'OPERACIONAL',
    STATUS_RAMPA    VARCHAR(20) DEFAULT 'OPERACIONAL',
    DATA_CADASTRO   TIMESTAMP   DEFAULT CURRENT_TIMESTAMP,
    CHECK (STATUS_ELEVADOR IN ('OPERACIONAL','ALERTA','QUEBRADO')),
    CHECK (STATUS_RAMPA    IN ('OPERACIONAL','ALERTA','QUEBRADO'))
);

CREATE TABLE TB_MECANICO (
    ID_MECANICO   INT AUTO_INCREMENT PRIMARY KEY,
    NOME          VARCHAR(100) NOT NULL,
    CPF           VARCHAR(14)  NOT NULL UNIQUE,
    ESPECIALIDADE VARCHAR(50)  NOT NULL
);

-- ID_MECANICO aceita NULL: a OS aberta automaticamente pelo sensor IoT
-- ainda nao tem responsavel; o gestor o atribui na triagem.
CREATE TABLE TB_ORDEM_SERVICO (
    ID_OS           INT AUTO_INCREMENT PRIMARY KEY,
    ID_VEICULO      INT          NOT NULL,
    ID_MECANICO     INT          NULL,
    TIPO_OS         VARCHAR(12)  NOT NULL DEFAULT 'CORRETIVA',
    DESCRICAO_FALHA TEXT         NOT NULL,
    DATA_ABERTURA   DATETIME     NOT NULL,
    DATA_AGENDADA   DATETIME     NULL,
    DATA_CONCLUSAO  DATETIME     NULL,
    STATUS_OS       VARCHAR(20)  DEFAULT 'EM_ABERTO',
    CHECK (TIPO_OS   IN ('CORRETIVA','PREVENTIVA')),
    CHECK (STATUS_OS IN ('EM_ABERTO','AGENDADA','CONCLUIDA')),
    FOREIGN KEY (ID_VEICULO)  REFERENCES TB_VEICULO(ID_VEICULO),
    FOREIGN KEY (ID_MECANICO) REFERENCES TB_MECANICO(ID_MECANICO)
);

CREATE INDEX IX_OS_STATUS ON TB_ORDEM_SERVICO (STATUS_OS, DATA_ABERTURA);
-- Consulta 1: ordens em aberto com veiculo e mecanico (INNER JOIN)
SELECT os.ID_OS, v.PLACA, v.MODELO, m.NOME AS MECANICO_RESPONSAVEL,
       os.DESCRICAO_FALHA, os.STATUS_OS, os.DATA_ABERTURA
FROM TB_ORDEM_SERVICO os
INNER JOIN TB_VEICULO  v ON os.ID_VEICULO  = v.ID_VEICULO
INNER JOIN TB_MECANICO m ON os.ID_MECANICO = m.ID_MECANICO
WHERE os.STATUS_OS = 'EM_ABERTO'
ORDER BY os.DATA_ABERTURA DESC;

-- Consulta 2 (RF05): historico completo de uma placa, inclusive OS sem mecanico (LEFT JOIN)
SELECT os.ID_OS, os.TIPO_OS, os.STATUS_OS, os.DATA_ABERTURA, os.DATA_CONCLUSAO,
       COALESCE(m.NOME, '(nao atribuido)') AS MECANICO
FROM TB_ORDEM_SERVICO os
INNER JOIN TB_VEICULO v ON os.ID_VEICULO = v.ID_VEICULO
LEFT  JOIN TB_MECANICO m ON os.ID_MECANICO = m.ID_MECANICO
WHERE v.PLACA = 'BRA2E19'
ORDER BY os.DATA_ABERTURA;

-- Consulta 3 (RF06): percentual da frota com acessibilidade 100% ativa
SELECT COUNT(*) AS TOTAL,
       SUM(CASE WHEN STATUS_ELEVADOR = 'OPERACIONAL' AND STATUS_RAMPA = 'OPERACIONAL'
                THEN 1 ELSE 0 END) AS ATIVOS,
       ROUND(100.0 * SUM(CASE WHEN STATUS_ELEVADOR = 'OPERACIONAL' AND STATUS_RAMPA = 'OPERACIONAL'
                              THEN 1 ELSE 0 END) / COUNT(*), 1) AS PERCENTUAL_ATIVO
FROM TB_VEICULO;

-- Fechamento de OS (transacao: OS concluida e veiculo liberado juntos)
START TRANSACTION;
UPDATE TB_ORDEM_SERVICO SET STATUS_OS = 'CONCLUIDA', DATA_CONCLUSAO = NOW() WHERE ID_OS = 1;
UPDATE TB_VEICULO SET STATUS_ELEVADOR = 'OPERACIONAL' WHERE ID_VEICULO = 1;
COMMIT;

Coleção telemetry_logs

  • GeoJSON: coordenadas na ordem [longitude, latitude], exigida pelo índice 2dsphere.
  • Índice composto { bus_id: 1, timestamp: -1 } para buscar a última leitura de cada ônibus.
  • TTL de 90 dias no campo timestamp: os dados brutos expiram sozinhos, como pede a LGPD.
  • Esquema flexível: firmware novo pode enviar sensores extras sem migração.
db.telemetry_logs.createIndex({ location: "2dsphere" })
db.telemetry_logs.createIndex({ bus_id: 1, timestamp: -1 })
db.telemetry_logs.createIndex({ timestamp: 1 },
  { expireAfterSeconds: 7776000 })

/* Grava em arquivo temporario e substitui o original: evita perda em caso de falha. */
int salvarFrota(const Frota *f) {
    const char *tmp = ARQ_FROTA ".tmp";
    FILE *arq = fopen(tmp, "w");
    int i;
    if (arq == NULL) {
        printf("  [ERRO] Nao foi possivel gravar o arquivo de frota.\n");
        return 0;
    }
    for (i = 0; i < f->quantidade; i++) {
        const Veiculo *v = &f->itens[i];
        fprintf(arq, "%d;%s;%s;%s;%d;%s;%s\n", v->id, v->placa, v->prefixo,
                v->modelo, v->ano, v->status_elevador, v->status_rampa);
    }
    if (fclose(arq) != 0) return 0;
    remove(ARQ_FROTA);
    if (rename(tmp, ARQ_FROTA) != 0) {
        printf("  [ERRO] Falha ao substituir o arquivo de frota.\n");
        return 0;
    }
    return 1;
}

            

          

        

      

    


    
    

      

        

          Seção 6 · PIM III
          
Direitos humanos, acessibilidade e LGPD

          
Tecnologia que existe para garantir o direito de ir e vir (art. 5º, XV, da Constituição) de quem mais depende do transporte público.

        

        

          

            WCAG 2.1 · Nível AA
            
Acessibilidade digital

            
    
              
  • Perceptível — contraste mínimo de 4,5:1, textos alternativos em todos os ícones.
    • 
              
  • Operável — navegação completa por teclado, sem armadilhas de foco, animações pausáveis.
    • 
              
  • Compreensível — linguagem simples, mensagens de erro que dizem como corrigir.
    • 
              
  • Robusto — HTML semântico e ARIA testados com TalkBack e VoiceOver.
    • 
              
  • Vídeos com janela de Libras e navegação por voz no aplicativo do cidadão.
    • 
            

          

          

            Lei nº 13.709/2018
            
LGPD aplicada à frota

            
    
              
  • Minimização — o app do cidadão recebe a posição do ônibus, nunca a do passageiro.
    • 
              
  • Anonimização — logs de sensores sem vínculo com o operador ou motorista.
    • 
              
  • Retenção — telemetria bruta expira em 90 dias (TTL no MongoDB).
    • 
              
  • Transparência — política de privacidade em linguagem simples e em Libras.
    • 
              
  • Segurança — TLS 1.3 em trânsito e controle de acesso por perfil (RF07).
    • 
            

          

          

            Lei nº 13.146/2015
            
Estatuto da Pessoa com Deficiência

            
    
              
  • Garante transporte acessível em igualdade de oportunidades com as demais pessoas.
    • 
              
  • O RF06 dá ao poder público um indicador objetivo: percentual da frota com acessibilidade 100% ativa.
    • 
              
  • A OS automática (RF03) reduz o tempo de um elevador quebrado em circulação.
    • 
              
  • Público atendido: PCDs, idosos, gestantes e pessoas com mobilidade reduzida.
    •
